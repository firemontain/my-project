## 一、模块定位

赌场经营模块负责管理赌场的**设施系统**、**NPC引流计算**、**经营事件处理**和**赌场规模评级**。赌场是玩家"延伸的身体"——每个设施都是玩家意志的物理化体现，经营层服务于牌局层，为牌局提供能力池、引流目标NPC、通过设施联动强化牌局表现。

### 核心职责

1. **设施管理**：维护设施列表、处理升级/降级、计算效果叠加
2. **NPC引流**：根据设施组合计算各类NPC的出现权重
3. **赌场评级**：根据设施总体状态计算赌场规模评级
4. **经营事件**：处理随机事件的触发和效果执行
5. **能力池联动**：设施变化时通知能力系统更新可用卡池

```gdscript
func get_facility_level(facility_id: String) -> int:

## 检查设施是否已解锁
func has_facility(facility_id: String) -> bool:
    return facilities.has(facility_id) and facilities[facility_id] > 0

## 检查设施是否达到最大等级
func is_facility_maxed(facility_id: String) -> bool:
    var data := get_facility_data(facility_id)
    if not data:
        return true
    return get_facility_level(facility_id) >= data.max_level

## 获取设施数据
func get_facility_data(facility_id: String) -> FacilityData:
    return _facility_data_cache.get(facility_id)

## 计算所有设施的解锁能力ID列表
func get_all_unlocked_ability_ids() -> Array[String]:
    var result: Array[String] = []
    for facility_id in facilities:
        var level := facilities[facility_id] as int
        var data := get_facility_data(facility_id)
        if data and data.unlocked_abilities.has(level):
            result.append_array(data.unlocked_abilities[level])
    return result

## 计算赌场评级
func calculate_rating() -> int:
    # 基于设施总等级、设施多样性、特殊设施加成计算
    var total_level: int = 0
    var facility_count: int = 0
    
    for facility_id in facilities:
        var level := facilities[facility_id] as int
        if level > 0:
            total_level += level
            facility_count += 1
    
    # 简化公式：评级 = 设施总等级 / 5，上限10
    casino_rating = mini(10, (total_level + facility_count * 2) / 5)
    return casino_rating
```

### 2.4 FacilityUpgrader 设施升级器

```gdscript
# gameplay/casino/FacilityUpgrader.gd
class_name FacilityUpgrader
extends RefCounted

## 升级结果枚举
enum UpgradeResult {
    SUCCESS,
    INSUFFICIENT_FUNDS,
    MAX_LEVEL_REACHED,
    FACILITY_NOT_FOUND,
    LOCKED_BY_REQUIREMENT
}

## 计算升级费用
## @param casino_state: 赌场状态
## @param facility_id: 设施ID
## @return: 升级费用，-1表示无法升级
func calculate_upgrade_cost(casino_state: CasinoState, facility_id: String) -> int:
    var data := casino_state.get_facility_data(facility_id)
    if not data:
        return -1
    
    var current_level := casino_state.get_facility_level(facility_id)
    if current_level >= data.max_level:
        return -1
    
    return data.upgrade_costs[current_level]

## 验证是否可以升级
func can_upgrade(
    casino_state: CasinoState,
    facility_id: String,
    available_funds: int
) -> UpgradeResult:
    var data := casino_state.get_facility_data(facility_id)
    if not data:
        return UpgradeResult.FACILITY_NOT_FOUND
    
    if casino_state.is_facility_maxed(facility_id):
        return UpgradeResult.MAX_LEVEL_REACHED
    
    var cost := calculate_upgrade_cost(casino_state, facility_id)
    if cost > available_funds:
        return UpgradeResult.INSUFFICIENT_FUNDS
    
    return UpgradeResult.SUCCESS

## 执行升级（仅修改状态，资金扣除由 EconomyManager 处理）
## @return: 新等级
func do_upgrade(casino_state: CasinoState, facility_id: String) -> int:
    var current_level := casino_state.get_facility_level(facility_id)
    var new_level := current_level + 1
    casino_state.facilities[facility_id] = new_level
    casino_state.calculate_rating()
    return new_level

## 执行降级
## @return: 新等级
func do_downgrade(casino_state: CasinoState, facility_id: String) -> int:
    var current_level := casino_state.get_facility_level(facility_id)
    if current_level <= 0:
        return 0
    
    var new_level := current_level - 1
    casino_state.facilities[facility_id] = new_level
    casino_state.calculate_rating()
    return new_level

## 随机选择设施进行降级（破产惩罚用）
## @param count: 需要降级的数量
## @return: 被降级的设施ID列表
func select_facilities_for_downgrade(casino_state: CasinoState, count: int) -> Array[String]:
    var downgradable: Array[String] = []
    for facility_id in casino_state.facilities:
        if casino_state.facilities[facility_id] > 0:
            downgradable.append(facility_id)
    
    downgradable.shuffle()
    return downgradable.slice(0, mini(count, downgradable.size()))
```

```gdscript
# gameplay/casino/NPCAttractionCalc.gd
class_name NPCAttractionCalc
extends RefCounted

## NPC层级枚举
enum NPCTier {
    COMMON,     # 普通：随机生成，自动进场
    ELITE,      # 精英：需满足设施条件
    SPECIAL,    # 特殊：叙事事件触发
    BOSS        # Boss：章节推进触发
}

## 计算各NPC的出现权重
## @param casino_state: 赌场状态
## @param npc_pool: 可用NPC ID列表
## @return: {npc_id: weight}
func calculate_attraction_weights(
    casino_state: CasinoState,
    npc_pool: Array[String]
) -> Dictionary:
    var weights: Dictionary = {}
    
    for npc_id in npc_pool:
        var npc_data := _get_npc_data(npc_id)
        if not npc_data:
            continue
        
        # 计算基础权重
        var base_weight := _get_base_weight(npc_data.npc_tier)
        
        # 计算设施吸引力加成
        var facility_bonus := _calculate_facility_bonus(casino_state, npc_data)
        
        # 检查出现条件
        if not _check_appearance_conditions(casino_state, npc_data):
            weights[npc_id] = 0
            continue
        
        weights[npc_id] = base_weight + facility_bonus
    
    return weights

## 检查精英NPC出现条件
func check_elite_conditions(casino_state: CasinoState, npc_data: NPCData) -> bool:
    # 检查赌场评级是否达标
    if casino_state.casino_rating < npc_data.required_rating:
        return false
    
    # 检查特定设施要求
    for facility_id in npc_data.required_facilities:
        var required_level: int = npc_data.required_facilities[facility_id]
        if casino_state.get_facility_level(facility_id) < required_level:
            return false
    
    return true

## 根据权重抽取进场NPC
## @param weights: 权重字典
## @param count: 抽取数量
## @return: 抽取的NPC ID列表
func draw_npcs(weights: Dictionary, count: int) -> Array[String]:
    var result: Array[String] = []
    var total_weight: float = 0.0
    
    for npc_id in weights:
        total_weight += weights[npc_id]
    
    for i in range(count):
        var roll := randf() * total_weight
        var cumulative: float = 0.0
        
        for npc_id in weights:
            cumulative += weights[npc_id]
            if roll <= cumulative:
                result.append(npc_id)
                break
    
    return result
```

## 四、设施-能力池联动规范

### 4.1 联动机制

设施与能力池的关系通过 **EventBus 信号** 实现解耦：

```
设施升级
    ↓
CasinoManager 发射 facility_upgraded(facility_id, new_level) 信号
    ↓
AbilityPoolManager 订阅信号
    ↓
根据新等级查询 FacilityData.unlocked_abilities
    ↓
将对应能力卡加入可用池
    ↓
发射 ability_pool_updated 信号
```
