## 一、模块定位

牌局引擎模块负责执行**完整的梭哈（五张种马）牌局逻辑**，包括牌堆管理、发牌流程、下注轮推进、手牌评估与摊牌结算。该模块是核心玩法的执行引擎，保障牌局规则执行一致性，支持完整的TDD测试覆盖。

### 核心职责

1. **牌堆管理**：洗牌、发牌、出千换牌操作
2. **牌局状态**：维护当前手牌、底池、轮次、行动窗口状态
3. **下注轮控制**：行动窗口开关、NPC决策调用时序、下注金额验证
4. **手牌评估**：完整的梭哈牌型判定与比较
5. **摊牌结算**：胜负判定与筹码分配

---

## 二、梭哈规则说明

### 2.1 五张种马结构

```
每人 1 张底牌（暗牌）+ 4 张明牌
共 4 轮下注，每轮下注前有行动窗口

流程：
发牌（1暗 + 1明）→ [行动窗口] → 第1轮下注
亮第3张 → [行动窗口] → 第2轮下注
亮第4张 → [行动窗口] → 第3轮下注
亮第5张 → [行动窗口] → 第4轮下注（最终轮）→ 摊牌
```

### 2.2 牌型大小（从大到小）

1. **皇家同花顺**：同花色 10-J-Q-K-A
2. **同花顺**：同花色连续五张
3. **四条**：四张相同点数
4. **葫芦**：三条 + 一对
5. **同花**：同花色任意五张
6. **顺子**：不同花色连续五张
7. **三条**：三张相同点数
8. **两对**：两组对子
9. **一对**：一组对子
10. **高牌**：无任何组合

### 2.3 AP（行动点）规则

- 每局开始获得 **4 AP**
- 不跨局积累，牌局结束后清零
- 能力卡消耗 1-2 AP

### 3.2 Deck 牌堆

```gdscript
# gameplay/poker/Deck.gd
class_name Deck
extends RefCounted

## 剩余牌堆
var _cards: Array[CardData] = []

## 已发出的牌
var _dealt_cards: Array[CardData] = []

## ── 初始化 ──

func _init() -> void:
    _build_standard_deck()
    shuffle()

## 构建标准52张牌
func _build_standard_deck() -> void:
    _cards.clear()
    for suit in CardData.Suit.values():
        for rank in range(CardData.Rank.TWO, CardData.Rank.ACE + 1):
            var card := CardData.new()
            card.suit = suit
            card.rank = rank
            _cards.append(card)

## ── 牌堆操作 ──

## 洗牌
func shuffle() -> void:
    _cards.shuffle()

## 发一张牌
func deal_card() -> CardData:
    if _cards.is_empty():
        return null
    var card := _cards.pop_front()
    _dealt_cards.append(card)
    return card

## 发多张牌
func deal_cards(count: int) -> Array[CardData]:
    var result: Array[CardData] = []
    for i in range(count):
        var card := deal_card()
        if card:
            result.append(card)
    return result

## 获取剩余牌数
func get_remaining_count() -> int:
    return _cards.size()

## ── 出千操作 ──

## 换牌（出千类能力）
## @param target_hand: 目标手牌数组
## @param card_index: 要替换的牌索引
## @param new_card: 新牌（从牌堆中指定或随机）
## @return: 被换出的牌
func swap_card(target_hand: Array[CardData], card_index: int, new_card: CardData = null) -> CardData:
    if card_index < 0 or card_index >= target_hand.size():
        return null
    
    var old_card := target_hand[card_index]
    
    # 如果没有指定新牌，从牌堆随机抽取
    if new_card == null:
        new_card = deal_card()
    else:
        # 从牌堆中移除指定的牌
        var idx := _cards.find(new_card)
        if idx >= 0:
            _cards.remove_at(idx)
            _dealt_cards.append(new_card)
    
    if new_card:
        target_hand[card_index] = new_card
    
    # 将旧牌放回牌堆底部
    _cards.append(old_card)
    
    return old_card

## 标记牌（出千类能力）
## 返回牌堆顶部若干张牌的信息
func peek_top_cards(count: int) -> Array[CardData]:
    var result: Array[CardData] = []
    for i in range(mini(count, _cards.size())):
        result.append(_cards[i])
    return result

## 重排牌堆顶部（出千类能力）
func reorder_top_cards(new_order: Array[CardData]) -> void:
    # 移除原位置的牌
    for card in new_order:
        var idx := _cards.find(card)
        if idx >= 0:
            _cards.remove_at(idx)
    
    # 按新顺序插入到牌堆顶部
    for i in range(new_order.size() - 1, -1, -1):
        _cards.insert(0, new_order[i])
```

### 3.3 HandEvaluator 手牌评估器

```gdscript
# gameplay/poker/HandEvaluator.gd
class_name HandEvaluator
extends RefCounted

## 牌型枚举
enum HandRank {
    HIGH_CARD = 0,          # 高牌
    ONE_PAIR = 1,           # 一对
    TWO_PAIR = 2,           # 两对
    THREE_OF_KIND = 3,      # 三条
    STRAIGHT = 4,           # 顺子
    FLUSH = 5,              # 同花
    FULL_HOUSE = 6,         # 葫芦
    FOUR_OF_KIND = 7,       # 四条
    STRAIGHT_FLUSH = 8,     # 同花顺
    ROYAL_FLUSH = 9         # 皇家同花顺
}

## 评估结果
class HandResult:
    extends RefCounted
    
    var rank: HandRank              # 牌型
    var primary_value: int = 0      # 主要比较值（如对子的点数）
    var secondary_value: int = 0    # 次要比较值（如两对的小对点数）
    var kickers: Array[int] = []    # 踢脚牌点数（用于同牌型比较）
    var description: String = ""    # 牌型描述

## ── 评估方法 ──

## 评估五张手牌
func evaluate(hand: Array[CardData]) -> HandResult:
    if hand.size() != 5:
        push_error("Hand must contain exactly 5 cards")
        return null
    
    var result := HandResult.new()
    
    # 排序手牌
    var sorted_hand := hand.duplicate()
    sorted_hand.sort_custom(_compare_by_rank)
    
    # 检查各牌型
    if _is_royal_flush(sorted_hand):
        result.rank = HandRank.ROYAL_FLUSH
        result.description = "皇家同花顺"
    elif _is_straight_flush(sorted_hand):
        result.rank = HandRank.STRAIGHT_FLUSH
        result.primary_value = _get_straight_high(sorted_hand)
        result.description = "同花顺"
    elif _is_four_of_kind(sorted_hand):
        result.rank = HandRank.FOUR_OF_KIND
        result.primary_value = _get_four_of_kind_value(sorted_hand)
        result.description = "四条"
    elif _is_full_house(sorted_hand):
        result.rank = HandRank.FULL_HOUSE
        var values := _get_full_house_values(sorted_hand)
        result.primary_value = values[0]
        result.secondary_value = values[1]
        result.description = "葫芦"
    elif _is_flush(sorted_hand):
        result.rank = HandRank.FLUSH
        result.kickers = _get_kickers(sorted_hand, 5)
        result.description = "同花"
    elif _is_straight(sorted_hand):
        result.rank = HandRank.STRAIGHT
        result.primary_value = _get_straight_high(sorted_hand)
        result.description = "顺子"
    elif _is_three_of_kind(sorted_hand):
        result.rank = HandRank.THREE_OF_KIND
        result.primary_value = _get_three_of_kind_value(sorted_hand)
        result.kickers = _get_kickers_excluding(sorted_hand, [result.primary_value], 2)
        result.description = "三条"
    elif _is_two_pair(sorted_hand):
        result.rank = HandRank.TWO_PAIR
        var pairs := _get_pairs(sorted_hand)
        result.primary_value = pairs[0]
        result.secondary_value = pairs[1]
        result.kickers = _get_kickers_excluding(sorted_hand, pairs, 1)
        result.description = "两对"
    elif _is_one_pair(sorted_hand):
        result.rank = HandRank.ONE_PAIR
        result.primary_value = _get_pair_value(sorted_hand)
        result.kickers = _get_kickers_excluding(sorted_hand, [result.primary_value], 3)
        result.description = "一对"
    else:
        result.rank = HandRank.HIGH_CARD
        result.kickers = _get_kickers(sorted_hand, 5)
        result.description = "高牌"
    
    return result

## 比较两手牌
## @return: 正数表示hand1赢，负数表示hand2赢，0表示平局
func compare(hand1: Array[CardData], hand2: Array[CardData]) -> int:
    var result1 := evaluate(hand1)
    var result2 := evaluate(hand2)
    
    return compare_results(result1, result2)

## 比较两个评估结果
func compare_results(result1: HandResult, result2: HandResult) -> int:
    # 先比较牌型
    if result1.rank != result2.rank:
        return result1.rank - result2.rank
    
    # 同牌型比较主要值
    if result1.primary_value != result2.primary_value:
        return result1.primary_value - result2.primary_value
    
    # 比较次要值
    if result1.secondary_value != result2.secondary_value:
        return result1.secondary_value - result2.secondary_value
    
    # 比较踢脚牌
    for i in range(mini(result1.kickers.size(), result2.kickers.size())):
        if result1.kickers[i] != result2.kickers[i]:
            return result1.kickers[i] - result2.kickers[i]
    
    return 0  # 完全相同

## ── 私有辅助方法 ──

func _compare_by_rank(a: CardData, b: CardData) -> bool:
    return a.rank > b.rank  # 降序排列

func _is_royal_flush(hand: Array[CardData]) -> bool:
    return _is_straight_flush(hand) and hand[0].rank == CardData.Rank.ACE

func _is_straight_flush(hand: Array[CardData]) -> bool:
    return _is_flush(hand) and _is_straight(hand)

func _is_four_of_kind(hand: Array[CardData]) -> bool:
    var counts := _get_rank_counts(hand)
    return counts.values().has(4)

func _is_full_house(hand: Array[CardData]) -> bool:
    var counts := _get_rank_counts(hand)
    return counts.values().has(3) and counts.values().has(2)

func _is_flush(hand: Array[CardData]) -> bool:
    var first_suit := hand[0].suit
    for card in hand:
        if card.suit != first_suit:
            return false
    return true

func _is_straight(hand: Array[CardData]) -> bool:
    var ranks: Array[int] = []
    for card in hand:
        ranks.append(card.rank)
    ranks.sort()
    
    # 检查普通顺子
    var is_normal := true
    for i in range(1, ranks.size()):
        if ranks[i] - ranks[i-1] != 1:
            is_normal = false
            break
    
    if is_normal:
        return true
    
    # 检查 A-2-3-4-5 小顺子
    if ranks == [2, 3, 4, 5, 14]:
        return true
    
    return false

func _is_three_of_kind(hand: Array[CardData]) -> bool:
    var counts := _get_rank_counts(hand)
    return counts.values().has(3) and not counts.values().has(2)

func _is_two_pair(hand: Array[CardData]) -> bool:
    var counts := _get_rank_counts(hand)
    var pair_count := 0
    for count in counts.values():
        if count == 2:
            pair_count += 1
    return pair_count == 2

func _is_one_pair(hand: Array[CardData]) -> bool:
    var counts := _get_rank_counts(hand)
    var pair_count := 0
    for count in counts.values():
        if count == 2:
            pair_count += 1
    return pair_count == 1

func _get_rank_counts(hand: Array[CardData]) -> Dictionary:
    var counts: Dictionary = {}
    for card in hand:
        if not counts.has(card.rank):
            counts[card.rank] = 0
        counts[card.rank] += 1
    return counts

func _get_straight_high(hand: Array[CardData]) -> int:
    var ranks: Array[int] = []
    for card in hand:
        ranks.append(card.rank)
    ranks.sort()
    
    # A-2-3-4-5 小顺子
    if ranks == [2, 3, 4, 5, 14]:
        return 5  # 5为最大
    
    return ranks[4]

func _get_four_of_kind_value(hand: Array[CardData]) -> int:
    var counts := _get_rank_counts(hand)
    for rank in counts:
        if counts[rank] == 4:
            return rank
    return 0

func _get_full_house_values(hand: Array[CardData]) -> Array[int]:
    var counts := _get_rank_counts(hand)
    var three_val := 0
    var two_val := 0
    for rank in counts:
        if counts[rank] == 3:
            three_val = rank
        elif counts[rank] == 2:
            two_val = rank
    return [three_val, two_val]

func _get_three_of_kind_value(hand: Array[CardData]) -> int:
    var counts := _get_rank_counts(hand)
    for rank in counts:
        if counts[rank] == 3:
            return rank
    return 0

func _get_pairs(hand: Array[CardData]) -> Array[int]:
    var counts := _get_rank_counts(hand)
    var pairs: Array[int] = []
    for rank in counts:
        if counts[rank] == 2:
            pairs.append(rank)
    pairs.sort()
    pairs.reverse()
    return pairs

func _get_pair_value(hand: Array[CardData]) -> int:
    var counts := _get_rank_counts(hand)
    for rank in counts:
        if counts[rank] == 2:
            return rank
    return 0

func _get_kickers(hand: Array[CardData], count: int) -> Array[int]:
    var result: Array[int] = []
    for card in hand:
        result.append(card.rank)
    result.sort()
    result.reverse()
    return result.slice(0, count)

func _get_kickers_excluding(hand: Array[CardData], exclude: Array[int], count: int) -> Array[int]:
    var result: Array[int] = []
    for card in hand:
        if not exclude.has(card.rank):
            result.append(card.rank)
    result.sort()
    result.reverse()
    return result.slice(0, count)
```

```gdscript
## 获取所有可见的牌（明牌）
func get_visible_cards() -> Array[CardData]:
    var result: Array[CardData] = []
    
    # 玩家的明牌
    for card in player_hand:
        if not card.is_face_down:
            result.append(card)
    
    # NPC的明牌
    for instance_id in npc_hands:
        for card in npc_hands[instance_id]:
            if not card.is_face_down:
                result.append(card)
    
    return result

## 获取玩家攻击性水平（供NPC决策参考）
func get_player_aggression_level() -> float:
    if betting_history.is_empty():
        return 0.5
    
    var raises := 0
    var total := 0
    for round_history in betting_history:
        for action in round_history:
            if action[2] == "player":
                total += 1
                if action[0] == "raise" or action[0] == "all_in":
                    raises += 1
    
    return float(raises) / float(total) if total > 0 else 0.5

## 获取存活的NPC数量
func get_active_npc_count() -> int:
    var count := 0
    for instance_id in npc_hands:
        var npc := NPCManager.get_instance(instance_id)
        if npc and not npc.has_folded:
            count += 1
    return count

func add_to_pot(amount: int) -> void:
    pot += amount

## 消耗行动点
func consume_ap(amount: int) -> bool:
    if ap_remaining >= amount:
        ap_remaining -= amount
        return true
    return false

## 记录行动
func record_action(action: String, amount: int, player_id: String) -> void:
    if betting_history.size() <= current_round:
        betting_history.append([])
    betting_history[current_round - 1].append([action, amount, player_id])

## 开始新轮次
func start_new_round() -> void:
    current_round += 1
    to_call = 0
    used_abilities_this_round.clear()
```

```gdscript
# gameplay/poker/BettingRoundCtrl.gd
class_name BettingRoundCtrl
extends RefCounted

## 轮次完成信号（内部使用）
signal round_completed(round: int)

## ── 配置 ──

## ── 初始化 ──

func setup_action_order(player_id: String, npc_instance_ids: Array[String]) -> void:
    action_order.clear()
    action_order.append(player_id)
    action_order.append_array(npc_instance_ids)

## ── 轮次控制 ──

## 开始下注轮
func start_betting_round(session_state: PokerSessionState) -> void:
    current_actor_index = 0
    needs_another_round = false
    
    # 第一轮需要盲注
    if session_state.current_round == 1:
        _post_blinds(session_state)
    
    # 发射行动窗口信号
    EventBus.betting_round_started.emit(session_state.current_round)
    EventBus.ability_window_opened.emit(session_state.current_round, session_state.ap_remaining)

## 关闭行动窗口，开始下注
func close_action_window(session_state: PokerSessionState) -> void:
    session_state.is_action_window_open = false
    _advance_to_next_actor(session_state)

## 处理玩家行动
func handle_player_action(
    session_state: PokerSessionState,
    action: String,
    amount: int
) -> bool:
    if action_order[current_actor_index] != "player":
        return false
    
    var success := _execute_action(session_state, "player", action, amount)
    if success:
        _advance_to_next_actor(session_state)
    return success

## 获取NPC行动
func get_npc_action(session_state: PokerSessionState) -> void:
    var actor_id := action_order[current_actor_index]
    if actor_id == "player":
        return
    
    var npc := NPCManager.get_instance(actor_id)
    if not npc or npc.has_folded:
        _advance_to_next_actor(session_state)
        return
    
    # 获取NPC决策
    var decision := npc.get_decision(session_state)
    
    # 执行决策
    var action := _convert_decision_to_action(decision)
    _execute_action(session_state, actor_id, action, decision.amount)
    
    _advance_to_next_actor(session_state)

## 检查轮次是否结束
func is_round_complete(session_state: PokerSessionState) -> bool:
    # 所有人都行动过且无需再轮
    return current_actor_index >= action_order.size() and not needs_another_round

## ── 私有方法 ──

func _post_blinds(session_state: PokerSessionState) -> void:
    session_state.to_call = big_blind
    session_state.min_raise = big_blind

func _execute_action(
    session_state: PokerSessionState,
    actor_id: String,
    action: String,
    amount: int
) -> bool:
    match action:
        "fold":
            _handle_fold(actor_id)
        "check":
            pass
        "call":
            _handle_call(session_state, actor_id, amount)
        "raise":
            _handle_raise(session_state, actor_id, amount)
        "all_in":
            _handle_all_in(session_state, actor_id, amount)
    
    session_state.record_action(action, amount, actor_id)
    return true

func _handle_fold(actor_id: String) -> void:
    if actor_id != "player":
        var npc := NPCManager.get_instance(actor_id)
        if npc:
            npc.has_folded = true

func _handle_call(session_state: PokerSessionState, actor_id: String, amount: int) -> void:
    session_state.add_to_pot(amount)
    if actor_id == "player":
        session_state.player_bet += amount
    else:
        session_state.npc_bets[actor_id] = session_state.npc_bets.get(actor_id, 0) + amount

func _handle_raise(session_state: PokerSessionState, actor_id: String, amount: int) -> void:
    session_state.add_to_pot(amount)
    session_state.to_call = amount
    session_state.min_raise = amount - session_state.to_call + amount
    needs_another_round = true
    
    if actor_id == "player":
        session_state.player_bet += amount
    else:
        session_state.npc_bets[actor_id] = session_state.npc_bets.get(actor_id, 0) + amount

func _handle_all_in(session_state: PokerSessionState, actor_id: String, amount: int) -> void:
    session_state.add_to_pot(amount)
    if amount > session_state.to_call:
        session_state.to_call = amount
        needs_another_round = true
    
    if actor_id == "player":
        session_state.player_bet += amount
    else:
        session_state.npc_bets[actor_id] = session_state.npc_bets.get(actor_id, 0) + amount
        var npc := NPCManager.get_instance(actor_id)
        if npc:
            npc.is_all_in = true

func _advance_to_next_actor(session_state: PokerSessionState) -> void:
    current_actor_index += 1
    
    if is_round_complete(session_state):
        round_completed.emit(session_state.current_round)
        return
    
    # 跳过已弃牌的NPC
    while current_actor_index < action_order.size():
        var actor_id := action_order[current_actor_index]
        if actor_id == "player":
            break
        var npc := NPCManager.get_instance(actor_id)
        if npc and not npc.has_folded:
            break
        current_actor_index += 1

func _convert_decision_to_action(decision: NPCDecisionResolver.NPCDecision) -> String:
    match decision.action:
        NPCDecisionResolver.NPCDecision.Action.FOLD:
            return "fold"
        NPCDecisionResolver.NPCDecision.Action.CHECK:
            return "check"
        NPCDecisionResolver.NPCDecision.Action.CALL:
            return "call"
        NPCDecisionResolver.NPCDecision.Action.RAISE:
            return "raise"
        NPCDecisionResolver.NPCDecision.Action.ALL_IN:
            return "all_in"
    return "fold"
```

### 3.6 ShowdownResolver 摊牌结算器

```gdscript
# gameplay/poker/ShowdownResolver.gd
class_name ShowdownResolver
extends RefCounted

## 结算结果
class ShowdownResult:
    extends RefCounted
    
    var winner_id: String           # 胜者ID（"player" 或 NPC instance_id）
    var winner_hand_rank: int       # 胜者牌型
    var winner_hand_desc: String    # 胜者牌型描述
    var pot_won: int                # 赢得的底池
    var all_hands: Dictionary       # 所有人的手牌评估结果

## 手牌评估器
var _evaluator: HandEvaluator = HandEvaluator.new()

## ── 结算方法 ──

## 执行摊牌结算
func resolve(session_state: PokerSessionState) -> ShowdownResult:
    var result := ShowdownResult.new()
    result.all_hands = {}
    
    # 收集所有存活参与者的手牌
    var participants: Dictionary = {}  # {id: hand}
    
    # 玩家手牌
    if session_state.player_hand.size() == 5:
        participants["player"] = session_state.player_hand
    
    # NPC手牌
    for instance_id in session_state.npc_hands:
        var npc := NPCManager.get_instance(instance_id)
        if npc and not npc.has_folded:
            participants[instance_id] = session_state.npc_hands[instance_id]
    
    # 如果只剩一人，直接获胜
    if participants.size() == 1:
        result.winner_id = participants.keys()[0]
        result.pot_won = session_state.pot
        return result
    
    # 评估所有手牌
    var best_id: String = ""
    var best_result: HandEvaluator.HandResult = null
    
    for id in participants:
        var hand_result := _evaluator.evaluate(participants[id])
        result.all_hands[id] = hand_result
        
        if best_result == null or _evaluator.compare_results(hand_result, best_result) > 0:
            best_id = id
            best_result = hand_result
    
    result.winner_id = best_id
    result.winner_hand_rank = best_result.rank
    result.winner_hand_desc = best_result.description
    result.pot_won = session_state.pot
    
    return result

## 处理结算结果
func process_result(result: ShowdownResult, session_state: PokerSessionState) -> void:
    # 更新资金
    if result.winner_id == "player":
        EconomyManager.add_funds(
            result.pot_won,
            EconomyTransaction.TransactionType.POKER_WIN,
            "poker_session"
        )
    else:
        var npc := NPCManager.get_instance(result.winner_id)
        if npc:
            npc.current_funds += result.pot_won
            npc.record_win()
    
    # 记录失败者
    for id in result.all_hands:
        if id != result.winner_id:
            if id != "player":
                var npc := NPCManager.get_instance(id)
                if npc:
                    npc.record_loss()
    
    # 发射结算信号
    EventBus.poker_session_ended.emit(result.winner_id, result.pot_won)
```

```gdscript
## 发初始牌（1暗 + 1明）
func _deal_initial_cards() -> void:
    # 玩家
    var player_hole := _session_state.deck.deal_card()
    player_hole.is_face_down = true
    var player_up := _session_state.deck.deal_card()
    _session_state.player_hand = [player_hole, player_up]
    
    # NPC
    for npc_id in _session_state.npc_hands:
        var npc_hole := _session_state.deck.deal_card()

    # 第2-4轮需要发新牌
    if _session_state.current_round > 1:
        _deal_round_cards()
    
    # 开启行动窗口
    _session_state.is_action_window_open = true
    _session_state.phase = PokerSessionState.SessionPhase.ACTION_WINDOW
    
    EventBus.ability_window_opened.emit(_session_state.current_round, _session_state.ap_remaining)

    _session_state.player_hand.append(player_card)
    
    # NPC

        var npc := NPCManager.get_instance(npc_id)
        if npc and not npc.has_folded:
            var npc_card := _session_state.deck.deal_card()
            _session_state.npc_hands[npc_id].append(npc_card)

func _on_round_completed(round: int) -> void:

    if _check_single_winner():
        return
    
    # 检查是否到达最后一轮
    if round >= 4:
        _do_showdown()
    else:
        _start_round()

## 检查是否只剩一个存活者
func _check_single_winner() -> bool:
    var survivors: Array[String] = ["player"]  # 假设玩家未弃牌
    
    for npc_id in _session_state.npc_hands:
        var npc := NPCManager.get_instance(npc_id)
        if npc and not npc.has_folded:
            survivors.append(npc_id)
    
    if survivors.size() == 1:
        end_session(survivors[0])
        return true
    
    return false

    _session_state.phase = PokerSessionState.SessionPhase.SHOWDOWN
    
    var result := _showdown_resolver.resolve(_session_state)

    _session_state.phase = PokerSessionState.SessionPhase.ENDED
    _is_session_active = false
```
