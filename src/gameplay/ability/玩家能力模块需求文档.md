## 一、模块定位

玩家能力模块负责管理**能力卡池**、**局内Build构建**、**AP（行动点）系统**和**能力效果执行**。该模块是连接经营层和牌局层的核心桥梁，将设施升级转化为牌局中的战术选项，实现"降维打击"的核心体验。

### 核心职责

1. **能力池管理**：根据设施等级维护当前可用能力卡池
2. **抽卡构筑**：开局抽卡、玩家选卡、构建局内Build
3. **AP系统**：管理行动点的分配与消耗
4. **效果执行**：将能力效果分发到各目标系统（通过EventBus）
5. **中途追加**：牌局中击败NPC后的追加选卡机会

### 2.2 PlayerBuild 局内构建

```gdscript
# gameplay/ability/PlayerBuild.gd
class_name PlayerBuild
extends RefCounted

## ── 已选能力卡 ──

## 本局携带的能力卡列表
var selected_cards: Array[AbilityCardData] = []

## 被动能力卡（自动生效）
var passive_cards: Array[AbilityCardData] = []

## ── 使用状态 ──

## 各卡牌剩余使用次数
var remaining_uses: Dictionary = {}  # {card_id: remaining}

## 各卡牌冷却剩余轮数
var cooldown_remaining: Dictionary = {}  # {card_id: rounds}

## ── 初始化 ──

## 从选定的卡牌构建Build
func build_from_selection(cards: Array[AbilityCardData]) -> void:
    selected_cards.clear()
    passive_cards.clear()
    remaining_uses.clear()
    cooldown_remaining.clear()
    
    for card in cards:
        if card.is_passive:
            passive_cards.append(card)
        else:
            selected_cards.append(card)
        
        # 初始化使用次数
        if card.uses_per_session > 0:
            remaining_uses[card.card_id] = card.uses_per_session

## ── 查询方法 ──

## 检查能力卡是否可用
func can_use(card_id: String, current_ap: int) -> bool:
    var card := _find_card(card_id)
    if not card:
        return false
    
    # 检查AP
    if current_ap < card.ap_cost:
        return false
    
    # 检查使用次数
    if remaining_uses.has(card_id) and remaining_uses[card_id] <= 0:
        return false
    
    # 检查冷却
    if cooldown_remaining.has(card_id) and cooldown_remaining[card_id] > 0:
        return false
    
    return true

## 获取所有可用能力卡
func get_available_cards(current_ap: int) -> Array[AbilityCardData]:
    var result: Array[AbilityCardData] = []
    for card in selected_cards:
        if can_use(card.card_id, current_ap):
            result.append(card)
    return result

## ── 使用能力 ──

## 消耗能力使用次数
func consume_use(card_id: String) -> void:
    if remaining_uses.has(card_id):
        remaining_uses[card_id] -= 1
    
    var card := _find_card(card_id)
    if card and card.cooldown_rounds > 0:
        cooldown_remaining[card_id] = card.cooldown_rounds

## ── 轮次更新 ──

## 每轮开始时更新冷却
func on_round_start() -> void:
    for card_id in cooldown_remaining:
        if cooldown_remaining[card_id] > 0:
            cooldown_remaining[card_id] -= 1

## ── 追加能力 ──

## 牌局中追加能力卡
func add_mid_session_card(card: AbilityCardData) -> void:
    if card.is_passive:
        passive_cards.append(card)
    else:
        selected_cards.append(card)
    
    if card.uses_per_session > 0:
        remaining_uses[card.card_id] = card.uses_per_session

## ── 私有方法 ──

func _find_card(card_id: String) -> AbilityCardData:
    for card in selected_cards:
        if card.card_id == card_id:
            return card
    return null
```

### 2.3 AbilityDraftSystem 抽卡系统

```gdscript
# gameplay/ability/AbilityDraftSystem.gd
class_name AbilityDraftSystem
extends RefCounted

## ── 配置 ──

## 普通抽卡数量
const NORMAL_DRAFT_COUNT: int = 5

## 高级抽卡数量（赌场评级高时）
const ENHANCED_DRAFT_COUNT: int = 7

## 选择数量
const SELECTION_COUNT: int = 3

## 中途追加抽卡数量
const MID_SESSION_DRAFT_COUNT: int = 2

## ── 状态 ──

## 当前候选卡牌
var current_candidates: Array[AbilityCardData] = []

## 已选择的卡牌
var current_selections: Array[AbilityCardData] = []

## ── 抽卡方法 ──

## 执行开局抽卡
## @param ability_pool: 当前可用能力池
## @param casino_rating: 赌场评级（影响抽卡数量）
## @return: 候选卡牌列表
func perform_draft(ability_pool: Array[AbilityCardData], casino_rating: int) -> Array[AbilityCardData]:
    current_candidates.clear()
    current_selections.clear()
    
    var draft_count := ENHANCED_DRAFT_COUNT if casino_rating >= 7 else NORMAL_DRAFT_COUNT
    
    # 从池中随机抽取
    var pool_copy := ability_pool.duplicate()
    pool_copy.shuffle()
    
    for i in range(mini(draft_count, pool_copy.size())):
        current_candidates.append(pool_copy[i])
    
    return current_candidates

## 执行中途追加抽卡
## @param ability_pool: 当前可用能力池
## @param count: 追加数量
## @return: 候选卡牌列表
func offer_mid_session_draft(ability_pool: Array[AbilityCardData], count: int) -> Array[AbilityCardData]:
    current_candidates.clear()
    
    # 排除已选择的卡牌
    var available: Array[AbilityCardData] = []
    for card in ability_pool:
        var already_selected := false
        for selected in current_selections:
            if selected.card_id == card.card_id:
                already_selected = true
                break
        if not already_selected:
            available.append(card)
    
    available.shuffle()
    
    for i in range(mini(count, available.size())):
        current_candidates.append(available[i])
    
    return current_candidates

## ── 选择方法 ──

## 选择一张卡牌
func select_card(card_id: String) -> bool:
    if current_selections.size() >= SELECTION_COUNT:
        return false
    
    for card in current_candidates:
        if card.card_id == card_id:
            current_selections.append(card)
            return true
    
    return false

## 取消选择
func deselect_card(card_id: String) -> bool:
    for i in range(current_selections.size()):
        if current_selections[i].card_id == card_id:
            current_selections.remove_at(i)
            return true
    return false

## 确认选择
func confirm_selection() -> Array[AbilityCardData]:
    if current_selections.size() < SELECTION_COUNT:
        push_warning("Not enough cards selected")
    return current_selections

## ── 查询方法 ──

func get_candidates() -> Array[AbilityCardData]:
    return current_candidates

func get_selections() -> Array[AbilityCardData]:
    return current_selections

func get_remaining_selection_count() -> int:
    return SELECTION_COUNT - current_selections.size()
```

### 2.4 AbilityEffectResolver 效果执行器

```gdscript
# gameplay/ability/AbilityEffectResolver.gd
class_name AbilityEffectResolver
extends RefCounted

## 效果执行结果
class EffectResult:
    extends RefCounted
    
    var success: bool = false
    var effect_data: Dictionary = {}
    var error_message: String = ""

## ── 效果执行 ──

## 执行能力效果
## @param card: 能力卡数据
## @param target_npc_id: 目标NPC ID（可选）
## @return: 执行结果
func execute_effect(card: AbilityCardData, target_npc_id: String = "") -> EffectResult:
    var result := EffectResult.new()
    
    match card.ability_type:
        AbilityCardData.AbilityType.INTEL:
            result = _execute_intel(card, target_npc_id)
        AbilityCardData.AbilityType.PHYSIOLOGICAL:
            result = _execute_physiological(card, target_npc_id)
        AbilityCardData.AbilityType.CHEATING:
            result = _execute_cheating(card)
        AbilityCardData.AbilityType.ENVIRONMENT:
            result = _execute_environment(card)
    
    # 通过 EventBus 广播能力使用
    # 注意：此处仅发射信号，具体执行由各 Manager 响应
    EventBus.ability_used.emit(card.card_id, card.ability_type, target_npc_id)
    
    return result

## ── 各类型效果实现 ──

## 情报类能力
func _execute_intel(card: AbilityCardData, target_npc_id: String) -> EffectResult:
    var result := EffectResult.new()
    
    # 情报类能力的具体执行由 NPCManager 响应信号后处理
    # 这里只负责收集需要的信息类型
    result.success = true
    result.effect_data = {
        "info_types": card.intel_info_types,
        "target_npc_id": target_npc_id
    }
    
    return result

## 生理攻击类能力
func _execute_physiological(card: AbilityCardData, target_npc_id: String) -> EffectResult:
    var result := EffectResult.new()
    
    # 生理攻击的具体执行由 NPCManager 响应信号后处理
    result.success = true
    result.effect_data = {
        "excitement_delta": card.excitement_delta,
        "stress_delta": card.stress_delta,
        "target_npc_id": target_npc_id,
        "affects_all": card.affects_all_npcs
    }
    
    return result

## 出千类能力
func _execute_cheating(card: AbilityCardData) -> EffectResult:
    var result := EffectResult.new()
    
    # 出千类的具体执行由 PokerSessionMgr 响应信号后处理
    result.success = true
    result.effect_data = {
        "swap_config": card.swap_card_config
    }
    
    return result

## 环境联动类能力
func _execute_environment(card: AbilityCardData) -> EffectResult:
    var result := EffectResult.new()
    
    # 环境联动由 NPCManager 批量处理
    result.success = true
    result.effect_data = {
        "excitement_delta": card.excitement_delta,
        "stress_delta": card.stress_delta,
        "affects_all": true
    }
    
    return result
```

signal draft_completed(selected_cards: Array[AbilityCardData])

    EventBus.facility_downgraded.connect(_on_facility_downgraded)

    # 获取所有设施解锁的能力ID
    var unlocked_ids := CasinoManager.get_all_unlocked_abilities()
    
    for ability_id in unlocked_ids:
        if _all_ability_data.has(ability_id):
            _ability_pool.append(_all_ability_data[ability_id])
    
    ability_pool_updated.emit(_ability_pool)

## 获取当前能力池
func get_ability_pool() -> Array[AbilityCardData]:
    return _ability_pool

## ── 抽卡流程 ──

## 开始抽卡阶段
func start_draft() -> Array[AbilityCardData]:
    var casino_rating := CasinoManager.casino_rating
    var candidates := _draft_system.perform_draft(_ability_pool, casino_rating)
    draft_started.emit(candidates)
    return candidates

## 选择卡牌
func select_card(card_id: String) -> bool:
    return _draft_system.select_card(card_id)

## 取消选择
func deselect_card(card_id: String) -> bool:
    return _draft_system.deselect_card(card_id)

## 确认选择，构建Build
func confirm_draft() -> PlayerBuild:
    var selected := _draft_system.confirm_selection()
    
    _current_build = PlayerBuild.new()
    _current_build.build_from_selection(selected)
    
    draft_completed.emit(selected)
    return _current_build

## 获取抽卡状态
func get_draft_candidates() -> Array[AbilityCardData]:
    return _draft_system.get_candidates()

func get_draft_selections() -> Array[AbilityCardData]:
    return _draft_system.get_selections()

func get_remaining_selections() -> int:
    return _draft_system.get_remaining_selection_count()

## ── 局内Build操作 ──

## 获取当前Build
func get_current_build() -> PlayerBuild:
    return _current_build

## 检查能力是否可用
func can_use_ability(card_id: String) -> bool:
    if not _current_build:
        return false
    return _current_build.can_use(card_id, PokerSessionMgr.ap_remaining)

## 使用能力
func use_ability(card_id: String, target_npc_id: String = "") -> AbilityEffectResolver.EffectResult:
    if not _current_build:
        var result := AbilityEffectResolver.EffectResult.new()
        result.error_message = "No active build"
        return result
    
    var card := _find_card_in_build(card_id)
    if not card:
        var result := AbilityEffectResolver.EffectResult.new()
        result.error_message = "Card not found in build"
        return result
    
    if not can_use_ability(card_id):
        var result := AbilityEffectResolver.EffectResult.new()
        result.error_message = "Cannot use this ability"
        return result

    var result := _effect_resolver.execute_effect(card, target_npc_id)
    
    if result.success:
        _current_build.consume_use(card_id)
    
    return result

## 轮次开始时更新Build状态
func on_round_start() -> void:
    if _current_build:
        _current_build.on_round_start()

## 清除当前Build（牌局结束时）

    _current_build = null

## ── 中途追加 ──

    var candidates := _draft_system.offer_mid_session_draft(_ability_pool, count)
    mid_session_draft_offered.emit(candidates)
    return candidates

## 确认追加选择
func confirm_mid_session_selection(card_id: String) -> bool:
    for card in _draft_system.get_candidates():

## 四、能力效果执行路径

### 4.1 跨模块通信规则

AbilityEffectResolver **不直接调用**其他模块的内部类，所有效果执行通过 EventBus 信号分发：

```
玩家使用能力
    ↓
AbilityPoolManager.use_ability()
    ↓
AbilityEffectResolver.execute_effect()
    ↓
EventBus.ability_used.emit(card_id, ability_type, target_npc_id)
    ↓
┌─────────────────────────────────────────────────────────────┐
│ 各 Manager 订阅 ability_used 信号，执行对应逻辑：            │
│                                                             │
│ • NPCManager: 处理情报类、生理攻击类、环境联动类            │
│   → 调用 NPCInstance.apply_physiological_modifier()         │
│   → 返回档案信息给 UI 展示                                  │
│                                                             │
│ • PokerSessionMgr: 处理出千类                               │
│   → 调用 Deck.swap_card()                                   │
│   → 调用 Deck.peek_top_cards()                              │
└─────────────────────────────────────────────────────────────┘
```

### 4.2 各能力类型执行路径

| 能力类型 | 信号接收者 | 执行逻辑 |
|----------|------------|----------|
| 情报类 | NPCManager | 返回目标NPC的档案信息（阈值、弱点等） |
| 生理攻击类 | NPCManager | 调用目标NPC的生理系统修改兴奋度/压力 |
| 出千类 | PokerSessionMgr | 调用Deck执行换牌/标记牌/重排牌堆 |
| 环境联动类 | NPCManager | 批量对当前桌所有NPC应用生理修正 |

## 五、四大流派设计

### 5.1 流派概述

| 流派 | 核心策略 | 依赖设施 | 典型能力卡 |
|------|---------|---------|-----------|
| **出千流** | 直接控制牌面 | 千数后台 | 换牌、标记牌、控制发牌 |
| **心理爆破流** | 快速轰炸生理系统 | 无特殊依赖 | 秒跟注、敲击桌面、眼神对视 |
| **情报控场流** | 信息碾压 | 情报网络、豪华包厢 | 偷看底牌、读取档案、推算牌型 |
| **环境宰客流** | 设施被动联动 | 多设施协同 | 服务生扰动、调暗灯光、播放音乐 |
