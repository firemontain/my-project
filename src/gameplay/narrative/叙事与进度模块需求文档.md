## 一、模块定位

叙事与进度模块负责管理**主线剧情状态机**、**NPC档案系统**、**对话触发**和**哈迪斯式碎片叙事**。该模块确保叙事反馈与玩法结果形成持续关联，让玩家在反复游玩中逐步揭开完整的故事全貌。

### 核心职责

1. **剧情状态管理**：维护已解锁的剧情碎片、章节对话进度
2. **NPC档案系统**：管理NPC档案的解锁、背景故事段落、弱点提示
3. **对话触发**：条件分支对话、变量插值、时机触发
4. **碎片叙事**：哈迪斯式渐进叙事，破产重置不丢失已解锁内容

### 2.10 UnlockCondition 解锁条件

```gdscript
# gameplay/narrative/UnlockCondition.gd
class_name UnlockCondition
extends Resource

enum ConditionType {
    ALWAYS,             # 总是满足
    CHAPTER_COMPLETED,  # 章节完成
    NPC_DEFEATED,       # 击败特定NPC
    NPC_DEFEATED_COUNT, # 击败NPC次数
    FRAGMENT_UNLOCKED,  # 碎片已解锁
    VARIABLE_CHECK,     # 变量检查
    FACILITY_LEVEL,     # 设施等级
    CASINO_RATING,      # 赌场评级
    AND,                # 且条件
    OR                  # 或条件
}

## 条件类型
@export var condition_type: ConditionType

## 条件参数
@export var parameters: Dictionary = {}

## 子条件（用于 AND/OR）
@export var sub_conditions: Array[UnlockCondition] = []

## 评估条件是否满足
func evaluate(story_state: StoryState, game_context: Dictionary = {}) -> bool:
    match condition_type:
        ConditionType.ALWAYS:
            return true
        ConditionType.CHAPTER_COMPLETED:
            return _check_chapter_completed(game_context)
        ConditionType.NPC_DEFEATED:
            return _check_npc_defeated(game_context)
        ConditionType.NPC_DEFEATED_COUNT:
            return _check_npc_defeat_count(story_state)
        ConditionType.FRAGMENT_UNLOCKED:
            return story_state.is_fragment_unlocked(parameters.get("fragment_id", ""))
        ConditionType.VARIABLE_CHECK:
            return _check_variable(story_state)
        ConditionType.FACILITY_LEVEL:
            return _check_facility_level(game_context)
        ConditionType.CASINO_RATING:
            return _check_casino_rating(game_context)
        ConditionType.AND:
            return _check_and(story_state, game_context)
        ConditionType.OR:
            return _check_or(story_state, game_context)
    return false

## 私有方法...
func _check_chapter_completed(context: Dictionary) -> bool:
    var chapter_id: String = parameters.get("chapter_id", "")
    var completed_chapters: Array = context.get("completed_chapters", [])
    return chapter_id in completed_chapters

func _check_npc_defeated(context: Dictionary) -> bool:
    var npc_id: String = parameters.get("npc_id", "")
    var defeated_npc: String = context.get("defeated_npc", "")
    return npc_id == defeated_npc

func _check_npc_defeat_count(story_state: StoryState) -> bool:
    var npc_id: String = parameters.get("npc_id", "")
    var required_count: int = parameters.get("count", 1)
    var current_count: int = story_state.get_variable("defeat_count_%s" % npc_id, 0)
    return current_count >= required_count

func _check_variable(story_state: StoryState) -> bool:
    var var_name: String = parameters.get("var_name", "")
    var operator: String = parameters.get("operator", "==")
    var value: Variant = parameters.get("value", null)
    var current: Variant = story_state.get_variable(var_name)
    
    match operator:
        "==": return current == value
        "!=": return current != value
        ">": return current > value
        ">=": return current >= value
        "<": return current < value
        "<=": return current <= value
    return false

func _check_facility_level(context: Dictionary) -> bool:
    var facility_id: String = parameters.get("facility_id", "")
    var required_level: int = parameters.get("level", 1)
    var facilities: Dictionary = context.get("facilities", {})
    return facilities.get(facility_id, 0) >= required_level

func _check_casino_rating(context: Dictionary) -> bool:
    var required_rating: int = parameters.get("rating", 1)
    var current_rating: int = context.get("casino_rating", 0)
    return current_rating >= required_rating

func _check_and(story_state: StoryState, context: Dictionary) -> bool:
    for sub in sub_conditions:
        if not sub.evaluate(story_state, context):
            return false
    return true

func _check_or(story_state: StoryState, context: Dictionary) -> bool:
    for sub in sub_conditions:
        if sub.evaluate(story_state, context):
            return true
    return false
```

### 3.1 NarrativeManager 单例

```gdscript
# autoloads/NarrativeManager.gd
extends Node

## ── 信号定义 ──
signal story_node_triggered(node_id: String)
signal npc_archive_unlocked(npc_id: String)
signal dialogue_started(dialogue_id: String)
signal dialogue_ended(dialogue_id: String)
signal fragment_unlocked(fragment_id: String)
signal weakness_hint_revealed(npc_id: String, hint: WeaknessHint)

## ── 状态持有 ──
var _story_state: StoryState = StoryState.new()
var _pending_dialogues: Array[DialogueData] = []
var _current_dialogue: DialogueData = null
var _current_node: DialogueNode = null

## 数据缓存
var _story_nodes: Dictionary = {}      # {node_id: StoryNodeData}
var _dialogues: Dictionary = {}        # {dialogue_id: DialogueData}
var _archives: Dictionary = {}         # {npc_id: NPCArchive}

## ── 初始化 ──

func _ready() -> void:
    # 订阅游戏事件
    EventBus.poker_session_ended.connect(_on_poker_session_ended)
    EventBus.chapter_completed.connect(_on_chapter_completed)
    EventBus.facility_upgraded.connect(_on_facility_upgraded)
    EventBus.npc_trap_triggered.connect(_on_npc_trap_triggered)
    EventBus.game_phase_changed.connect(_on_game_phase_changed)
    
    _load_narrative_data()

## ── 故事节点管理 ──

## 检查并触发符合条件的故事节点
func check_and_trigger_nodes(trigger_type: StoryNodeData.TriggerType, context: Dictionary = {}) -> void:
    for node_id in _story_nodes:
        var node: StoryNodeData = _story_nodes[node_id]
        
        # 检查触发类型
        if node.trigger_type != trigger_type:
            continue
        
        # 检查是否已触发
        if _story_state.is_node_triggered(node_id):
            continue
        
        # 检查前置条件
        if not _check_prerequisites(node):
            continue
        
        # 检查触发条件
        if not node.trigger_condition.evaluate(_story_state, context):
            continue
        
        # 触发节点
        _trigger_node(node)

## 触发故事节点
func _trigger_node(node: StoryNodeData) -> void:
    _story_state.trigger_node(node.node_id)
    
    # 解锁剧情碎片
    for fragment_id in node.unlock_fragments:
        _story_state.unlock_fragment(fragment_id)
        fragment_unlocked.emit(fragment_id)
    
    # 解锁档案
    for npc_id in node.unlock_archives:
        _advance_archive(npc_id)
    
    # 设置变量
    for var_name in node.set_variables:
        _story_state.set_variable(var_name, node.set_variables[var_name])
    
    # 触发对话
    if not node.dialogue_id.is_empty():
        queue_dialogue(node.dialogue_id)
    
    story_node_triggered.emit(node.node_id)

func _check_prerequisites(node: StoryNodeData) -> bool:
    # 检查前置节点
    for prereq_id in node.prerequisite_nodes:
        if not _story_state.is_node_triggered(prereq_id):
            return false
    
    # 检查互斥节点
    for excluded_id in node.excluded_nodes:
        if _story_state.is_node_triggered(excluded_id):
            return false
    
    return true

## ── NPC档案管理 ──

## 推进NPC档案
func _advance_archive(npc_id: String) -> void:
    var new_progress := _story_state.advance_archive(npc_id)
    
    var archive: NPCArchive = _archives.get(npc_id)
    if not archive:
        return
    
    # 检查是否有新的弱点提示可解锁
    for hint in archive.weakness_hints:
        if hint.required_archive_progress == new_progress:
            weakness_hint_revealed.emit(npc_id, hint)
    
    npc_archive_unlocked.emit(npc_id)

## 获取NPC档案
func get_archive(npc_id: String) -> NPCArchive:
    return _archives.get(npc_id)

## 获取已解锁的档案段落
func get_unlocked_paragraphs(npc_id: String) -> Array[ArchiveParagraph]:
    var archive := get_archive(npc_id)
    if not archive:
        return []
    
    var progress := _story_state.get_archive_progress(npc_id)
    var result: Array[ArchiveParagraph] = []
    
    for i in range(mini(progress, archive.story_paragraphs.size())):
        result.append(archive.story_paragraphs[i])
    
    return result

## 获取已解锁的弱点提示
func get_unlocked_hints(npc_id: String) -> Array[WeaknessHint]:
    var archive := get_archive(npc_id)
    if not archive:
        return []
    
    var progress := _story_state.get_archive_progress(npc_id)
    var result: Array[WeaknessHint] = []
    
    for hint in archive.weakness_hints:
        if hint.required_archive_progress <= progress:
            result.append(hint)
    
    return result

## ── 对话系统 ──

## 将对话加入队列
func queue_dialogue(dialogue_id: String) -> void:
    var dialogue: DialogueData = _dialogues.get(dialogue_id)
    if dialogue:
        _pending_dialogues.append(dialogue)

## 开始下一个待处理对话
func start_next_dialogue() -> bool:
    if _pending_dialogues.is_empty():
        return false
    
    _current_dialogue = _pending_dialogues.pop_front()
    _current_node = _get_node(_current_dialogue.start_node_id)
    
    dialogue_started.emit(_current_dialogue.dialogue_id)
    return true

## 获取当前对话节点
func get_current_node() -> DialogueNode:
    return _current_node

## 处理文本显示后前进
func advance_dialogue() -> void:
    if not _current_node:
        return
    
    if _current_node.node_type == DialogueNode.NodeType.TEXT:
        _current_node = _get_node(_current_node.next_node_id)
        _process_current_node()

## 处理玩家选择
func make_choice(choice_index: int) -> void:
    if not _current_node or _current_node.node_type != DialogueNode.NodeType.CHOICE:
        return
    
    var choice: DialogueChoice = _current_node.choices[choice_index]
    
    # 设置变量
    for var_name in choice.set_variables:
        _story_state.set_variable(var_name, choice.set_variables[var_name])
    
    # 执行动作
    if choice.action:
        _execute_action(choice.action)
    
    # 跳转到下一节点
    _current_node = _get_node(choice.next_node_id)
    _process_current_node()

## 处理当前节点
func _process_current_node() -> void:
    if not _current_node:
        _end_dialogue()
        return
    
    match _current_node.node_type:
        DialogueNode.NodeType.BRANCH:
            _process_branch()
        DialogueNode.NodeType.ACTION:
            _process_action()
        DialogueNode.NodeType.END:
            _end_dialogue()

func _process_branch() -> void:
    var context := _build_game_context()
    
    for branch in _current_node.branch_conditions:
        if branch.condition.evaluate(_story_state, context):
            _current_node = _get_node(branch.next_node_id)
            _process_current_node()
            return
    
    # 默认分支
    _current_node = _get_node(_current_node.default_branch_id)
    _process_current_node()

func _process_action() -> void:
    _execute_action(_current_node.action)
    _current_node = _get_node(_current_node.next_node_id)
    _process_current_node()

func _execute_action(action: DialogueAction) -> void:
    match action.action_type:
        DialogueAction.ActionType.SET_VARIABLE:
            var var_name: String = action.parameters.get("var_name", "")
            var value: Variant = action.parameters.get("value")
            _story_state.set_variable(var_name, value)
        DialogueAction.ActionType.UNLOCK_FRAGMENT:
            var fragment_id: String = action.parameters.get("fragment_id", "")
            _story_state.unlock_fragment(fragment_id)
            fragment_unlocked.emit(fragment_id)
        DialogueAction.ActionType.UNLOCK_ARCHIVE:
            var npc_id: String = action.parameters.get("npc_id", "")
            _advance_archive(npc_id)
        DialogueAction.ActionType.GIVE_REWARD:
            var amount: int = action.parameters.get("amount", 0)
            EconomyManager.add_funds(amount, EconomyTransaction.TransactionType.EVENT_REWARD, "dialogue")

func _end_dialogue() -> void:
    if _current_dialogue and _current_dialogue.one_time_only:
        _story_state.complete_dialogue(_current_dialogue.dialogue_id)
    
    var dialogue_id := _current_dialogue.dialogue_id if _current_dialogue else ""
    _current_dialogue = null
    _current_node = null
    
    dialogue_ended.emit(dialogue_id)

## 解析对话文本中的变量
func resolve_text(text: String) -> String:
    var result := text
    
    # 查找所有 {var_name} 模式
    var regex := RegEx.new()
    regex.compile("\\{(\\w+)\\}")
    
    for match in regex.search_all(text):
        var var_name := match.get_string(1)
        var value := _story_state.get_variable(var_name, "???")
        result = result.replace("{%s}" % var_name, str(value))
    
    return result
```

## 四、哈迪斯式碎片叙事机制

### 4.1 核心规则

1. **渐进揭示**：每次击败精英/特殊NPC → 解锁其档案的下一段故事
2. **永久保留**：已触发的剧情碎片在破产重置后仍保留
3. **不重复**：玩家不会重复看到相同的叙事内容
4. **条件分支**：对话可根据玩家行为有不同发展

### 4.2 档案解锁流程

```
击败精英/特殊/Boss NPC
    ↓
NarrativeManager._on_poker_session_ended()
    ↓
检查NPC层级 >= ELITE?
    ├── 是 → _advance_archive(npc_id)
    │         ├── 更新 npc_archive_progress
    │         ├── 检查新弱点提示
    │         └── 发射 npc_archive_unlocked 信号
    └── 否 → 跳过
    ↓
UI 展示新解锁内容
```

### 4.3 对话触发流程

```
游戏事件发生（击败NPC、章节开始等）
    ↓
NarrativeManager.check_and_trigger_nodes()
    ↓
遍历所有故事节点
    ├── 检查触发类型匹配
    ├── 检查是否已触发
    ├── 检查前置/互斥条件
    └── 检查触发条件
    ↓
满足条件的节点 → _trigger_node()
    ├── 解锁碎片
    ├── 推进档案
    ├── 设置变量
    └── 加入对话队列
    ↓
UIManager 在合适时机调用 start_next_dialogue()
```
